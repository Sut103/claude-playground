## Technical Details

**更新の単一経路** — 全ての入力イベントを 1 つの `update()` に集約する。ビューごとに評価を呼ぶと 3 倍のコストになり、NFR-1.1（100ms）を無駄に圧迫する。

```js
(function (RT) {
  var els = {};   // DOM 参照をキャッシュ
  var saveTimer = null;

  function collectFlags() {
    var flags = "";
    ["g", "i", "m", "s", "u", "y"].forEach(function (f) {
      if (els["flag_" + f].checked) flags += f;
    });
    return flags;
  }

  function update() {
    var pattern = els.pattern.value;
    var flags = collectFlags();
    var text = els.testText.value;
    var replacement = els.replacement.value;

    els.patternDisplay.textContent = "/" + pattern + "/" + flags;

    var result = RT.evaluate(pattern, flags, text);   // 1 回だけ

    if (!result.ok) {
      els.error.textContent = result.error;
      RT.renderHighlight(els.highlight, [{ kind: "plain", text: text, matchIndex: null }]);
      RT.renderGroups(els.groups, result);
      RT.renderReplace(els.replaceOutput, pattern, flags, text, replacement);
      els.matchCount.textContent = "";
      els.truncatedNotice.textContent = "";
      return;
    }

    els.error.textContent = "";
    RT.renderHighlight(els.highlight, RT.buildSegments(text, result.matches));
    RT.renderGroups(els.groups, result);
    RT.renderReplace(els.replaceOutput, pattern, flags, text, replacement);
    els.matchCount.textContent = result.matches.length === 0
      ? "マッチなし"
      : result.matches.length + " 件";
    els.truncatedNotice.textContent = result.truncated
      ? "10,000 件で打ち切りました" : "";

    scheduleSave({ pattern: pattern, flags: flags, text: text, replacement: replacement });
  }

  function scheduleSave(state) {
    if (saveTimer) clearTimeout(saveTimer);
    saveTimer = setTimeout(function () { RT.saveState(state); }, 300);
  }
})(window.RT);
```

**エラー時の扱いに注意** — 上のコードはエラー時もハイライト領域にテスト文字列を `plain` で描いている。「画面が真っ白にならない」（US-4）を満たすため。グループ表と置換結果は各モジュールが `result.ok === false` を見て空状態にする（005 / 006 の AC）。

**起動シーケンス**

```js
function init() {
  cacheElements();
  var state = RT.loadState();
  els.pattern.value = state.pattern;
  els.testText.value = state.text;
  els.replacement.value = state.replacement;
  ["g","i","m","s","u","y"].forEach(function (f) {
    els["flag_" + f].checked = state.flags.indexOf(f) !== -1;
  });
  bindEvents();
  update();                       // 復元直後に描画（US-6）
}
```

**注意点**

- `#pattern-display` は `textContent` で組む。パターン文字列がそのまま入るため `innerHTML` は使わない
- debounce は保存のみに適用する。描画を debounce すると「即時更新」の受け入れ条件を満たせなくなる
- `#reset` は `RT.clearState()` の後に `RT.loadState()` を呼んで初期サンプルを入力欄に流し直し、`update()` する
- モジュール間でデータ形の不整合が出た場合、原則は該当モジュールへの差し戻し。`app.js` で変換して吸収すると、契約が 2 つ存在することになり後から読めなくなる
